{{- /* apply an overlay to given image parameters: DICT - image: the image to overlay - overlay: the image used to overlay - size: the target size of the result image - type: initialize values from params.huggingsite.images.type */ -}} {{- $opts := . }} {{- with $opts.type }} {{- with index site.Params.huggingsite.images . }} {{- $opts = merge . $opts }} {{- else }} {{- errorf "unknown site image definition '%s'" . }} {{- end }} {{- end }} {{- with $image := resources.Get $opts.image }} {{- $imageSize := or $opts.size $image.Width }} {{- with $image = .Resize (printf "%dx" $imageSize) }} {{- with resources.Get $opts.overlay }} {{- $tileSize := div $imageSize 2 }} {{- with .Resize (printf "%dx" $tileSize) }} {{ with images.Overlay . 0 $tileSize }} {{- return $image.Filter . }} {{- else }} {{- errorf "apply overlay failed for: [ %q ]" $opts.image }} {{- end }} {{- else }} {{- errorf "resize overlay image failed: [ %q ]" $opts.overlay }} {{ end }} {{- else }} {{- warnf "overlay image not found [ %q ]" $opts.overlay }} {{- end }} {{- return . }} {{- else }} {{- errorf "image resize failed [ %q ]" $opts.image }} {{- end }} {{- else }} {{- errorf "image not found [ %q ]" $opts.image }} {{- end }} {{- /**/ -}}
